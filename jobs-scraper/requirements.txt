# Playwright mueve los scrapers propios. Necesita un Chromium: en el VPS
# (Linux ARM64) Playwright NO publica binario, asi que la imagen instala el
# Chromium del sistema y se apunta ahi con SCRAPER_CHROMIUM (ver Dockerfile
# y panchito/sources/base.py::lanzar_navegador).
playwright==1.49.1

# Scrapling (https://github.com/D4Vinci/Scrapling) — mueve la fuente
# trabajo.org y aporta lo que Playwright NO cubre: la huella TLS/HTTP2.
# Un GET con User-Agent de Chrome miente en la cabecera pero saluda en TLS
# como Python, y esa contradiccion es justo lo que mira un WAF.
scrapling==0.4.15

# Las dependencias del extra `scrapling[fetchers]` van SUELTAS a proposito.
# Ese extra declara `playwright>=1.62.0`, y subir Playwright aca es
# arriesgado: en el VPS (ARM64) usamos el Chromium DEL SISTEMA, y un
# Playwright mas nuevo contra un Chromium mas viejo desincroniza el
# protocolo y tumba los tres scrapers que hoy funcionan. Verificado el
# 2026-09-03 que Scrapling 0.4.15 corre bien con playwright 1.49.1: la
# unica razon del `>=1.62` es su modo navegador, que esta fuente no usa.
curl_cffi>=0.16.1
